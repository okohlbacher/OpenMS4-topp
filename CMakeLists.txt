cmake_minimum_required(VERSION 3.24)
project(OpenMSTOPP VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(OPENMS4_WARNINGS_AS_ERRORS "Treat warnings in TOPP sources as errors" OFF)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
include(cmake/OpenMS4Dependencies.cmake)
openms4_find_core(COMPONENTS TestSupport)
openms4_find_package(OpenMSCLI)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
include(cmake/OpenMS4Tools.cmake)
openms4_load_tools(package_tools FeatureLinkerBase)
foreach(tool IN LISTS package_tools)
  add_executable(${tool} src/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE OpenMS::CLI OpenMS::Core)
  if(OPENMS4_WARNINGS_AS_ERRORS)
    if(MSVC)
      # Match the warning set enforced by -Wall/-Wextra on GCC and Clang. MSVC's
      # /W4 additionally reports legacy implicit conversions, shadowing, CRT
      # portability notices, and C4505 from imported Core headers.
      target_compile_options(${tool} PRIVATE /W4 /WX
        /wd4244 /wd4245 /wd4267 /wd4456 /wd4458 /wd4505 /wd4996)
    else()
      target_compile_options(${tool} PRIVATE -Wall -Wextra -Wpedantic -Werror)
      if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # Boost.Iostreams in Intel Homebrew instantiates deprecated implicit
        # copy constructors from its headers under current Apple Clang.
        target_compile_options(${tool} PRIVATE -Wno-deprecated-copy)
      endif()
    endif()
  endif()
endforeach()
target_link_libraries(FuzzyDiff PRIVATE OpenMS::TestFramework)
# OpenMSInfo reports which package build it belongs to, next to the Core SDK it links.
openms4_source_revision(package_revision package_dirty)
target_compile_definitions(OpenMSInfo PRIVATE
  OPENMS4_PACKAGE_REVISION="${package_revision}" OPENMS4_PACKAGE_DIRTY=$<BOOL:${package_dirty}>)
foreach(tool FileMerger DecoyDatabase)
  target_link_libraries(${tool} PRIVATE Boost::regex)
endforeach()
foreach(tool IsobaricWorkflow MetaProSIP)
  target_link_libraries(${tool} PRIVATE Eigen3::Eigen)
endforeach()
openms4_install_tools(topp bin ${package_tools})
option(OPENMS4_REGRESSION_TESTS "Enable installed fixture regression tests" OFF)
if(OPENMS4_REGRESSION_TESTS)
  openms4_find_package(OpenMSTestData)
  message(STATUS "Full numerical regression suite is available from OpenMSTestData; run the suite-level installed-tools harness")
endif()

if(BUILD_TESTING AND OPENMS4_REGRESSION_TESTS)
  find_program(PERCOLATOR_BINARY percolator)
  if(PERCOLATOR_BINARY)
    set(CF_OPENMS_TEST_DATA_PATH "${OpenMSTestData_TOPP_DIR}/")
    configure_file(tests/test_config.h.in generated/OpenMS/test_config.h @ONLY)
    add_executable(PercolatorAdapter_parity_test tests/PercolatorAdapter_parity_test.cpp ${OpenMS_TEST_SUPPORT_SOURCE})
    target_include_directories(PercolatorAdapter_parity_test PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(PercolatorAdapter_parity_test PRIVATE OpenMS::Core OpenMS::TestFramework)
    add_test(NAME PercolatorAdapter_parity COMMAND PercolatorAdapter_parity_test)
    set_tests_properties(PercolatorAdapter_parity PROPERTIES ENVIRONMENT
      "PERCOLATOR_BINARY=${PERCOLATOR_BINARY};PERCOLATOR_ADAPTER_BINARY=$<TARGET_FILE:PercolatorAdapter>")
  else()
    message(STATUS "PercolatorAdapter parity test not registered: external percolator unavailable")
  endif()
endif()
