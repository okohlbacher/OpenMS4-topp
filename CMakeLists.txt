cmake_minimum_required(VERSION 3.24)
project(OpenMSTOPP VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
include(cmake/OpenMS4Dependencies.cmake)
openms4_find_core()
openms4_find_package(OpenMSCLI)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
include(executables.cmake)
if(NOT OpenMS_WITH_WNETALIGN)
  list(REMOVE_ITEM package_tools FeatureLinkerWNet)
endif()
openms4_find_core(COMPONENTS TestSupport)
find_package(Boost 1.81 REQUIRED COMPONENTS regex)
find_package(Eigen3 3.4 REQUIRED CONFIG)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tools.json" metadata)
set(manifest "# name\tcategory\tversion\texecutable\n")
foreach(tool IN LISTS package_tools)
  add_executable(${tool} src/${tool}.cpp)
  target_compile_features(${tool} PRIVATE cxx_std_23)
  target_link_libraries(${tool} PRIVATE OpenMS::CLI OpenMS::Core)
  install(TARGETS ${tool} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  string(JSON count LENGTH "${metadata}" tools)
  math(EXPR last "${count} - 1")
  foreach(index RANGE ${last})
    string(JSON name GET "${metadata}" tools ${index} name)
    if(name STREQUAL tool)
      string(JSON category GET "${metadata}" tools ${index} category)
    endif()
  endforeach()
  string(APPEND manifest "${tool}\t${category}\t${PROJECT_VERSION}\t${CMAKE_INSTALL_BINDIR}/${tool}${CMAKE_EXECUTABLE_SUFFIX}\n")
  if(BUILD_TESTING AND NOT tool STREQUAL "OpenMSInfo")
    add_test(NAME ${tool}_write_ini COMMAND ${tool} -test -write_ini ${CMAKE_CURRENT_BINARY_DIR}/${tool}.ini)
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/ctd)
    add_test(NAME ${tool}_write_ctd COMMAND ${tool} -test -write_ctd ${CMAKE_CURRENT_BINARY_DIR}/ctd)
    set_tests_properties(${tool}_write_ini ${tool}_write_ctd PROPERTIES
      ENVIRONMENT "OPENMS_TOOL_PREFIX_PATH=${CMAKE_BINARY_DIR};OPENMS_DISABLE_UPDATE_CHECK=ON")
  endif()
endforeach()
target_link_libraries(FuzzyDiff PRIVATE OpenMS::TestFramework)
foreach(tool FileMerger DecoyDatabase NucleicAcidSearchEngine)
  target_link_libraries(${tool} PRIVATE Boost::regex)
endforeach()
foreach(tool IsobaricWorkflow MetaProSIP)
  target_link_libraries(${tool} PRIVATE Eigen3::Eigen)
endforeach()
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/share/openms4/tools)
string(REPLACE "\t${CMAKE_INSTALL_BINDIR}/" "\tbin/" build_manifest "${manifest}")
file(WRITE ${CMAKE_BINARY_DIR}/share/openms4/tools/topp.tools.tsv "${build_manifest}")
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/install-manifests)
file(WRITE ${CMAKE_BINARY_DIR}/install-manifests/topp.tools.tsv "${manifest}")
install(FILES ${CMAKE_BINARY_DIR}/install-manifests/topp.tools.tsv DESTINATION share/openms4/tools)
option(OPENMS4_REGRESSION_TESTS "Enable installed fixture regression tests" OFF)
if(OPENMS4_REGRESSION_TESTS)
  openms4_find_package(OpenMSTestData)
  message(STATUS "Full numerical regression suite is available from OpenMSTestData; run the suite-level installed-tools harness")
endif()

if(BUILD_TESTING AND OPENMS4_REGRESSION_TESTS)
  find_program(PERCOLATOR_BINARY percolator)
  if(PERCOLATOR_BINARY)
    set(CF_OPENMS_TEST_DATA_PATH "${OpenMSTestData_TOPP_DIR}/")
    configure_file(tests/test_config.h.in generated/OpenMS/test_config.h @ONLY)
    add_executable(PercolatorAdapter_parity_test tests/PercolatorAdapter_parity_test.cpp ${OpenMS_TEST_SUPPORT_SOURCE})
    target_include_directories(PercolatorAdapter_parity_test PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(PercolatorAdapter_parity_test PRIVATE OpenMS::Core OpenMS::TestFramework)
    add_test(NAME PercolatorAdapter_parity COMMAND PercolatorAdapter_parity_test)
    set_tests_properties(PercolatorAdapter_parity PROPERTIES ENVIRONMENT
      "PERCOLATOR_BINARY=${PERCOLATOR_BINARY};PERCOLATOR_ADAPTER_BINARY=$<TARGET_FILE:PercolatorAdapter>")
  else()
    message(STATUS "PercolatorAdapter parity test not registered: external percolator unavailable")
  endif()
endif()
